double stage_two(const tuner_trace::row_trace_t& trace, const double* p)
{
  const auto& features = tuner_trace::feature_table().features;
  const double mg_weight = trace.phase / static_cast<double>(GAME_PHASE_MAX);
  const double eg_weight = 1.0 - mg_weight;
  double sum = 0.0;
  for (const tuner_trace::entry_t& e : trace.entries) {
    const tuner_trace::feature_t& f = features[e.feature];
    if (!tuner_trace::clamped_site(f.site)) { continue; }
    sum += e.coef * (p[f.mg] * mg_weight + p[f.eg] * eg_weight);
  }
  double x[2][2] = {{0.0, 0.0}, {0.0, 0.0}};
  for (const tuner_trace::side_entry_t& e : trace.side_entries) {
    if (e.group != 0) { continue; }
    x[e.side][0] += e.coef * p[KSC_MG_BASE + e.input];
    x[e.side][1] += e.coef * p[KSC_EG_BASE + e.input];
  }
  for (int side = 0; side < 2; ++side) {
    const double own = (side == 0) ? -1.0 : 1.0;
    const double m = std::max(x[side][0], 0.0);
    const double n = std::max(x[side][1], 0.0);
    sum += own * (m * m * mg_weight + n * n * eg_weight) / KS_CURVE_DIVISOR;
  }
  return sum;
}

std::string with_side_to_move(const std::string& fen, char side)
{
  std::string flipped = fen;
  flipped[fen.find(' ') + 1] = side;
  return flipped;
}

bool loads_either_side(const std::string& fen)
{
  return load_FEN(with_side_to_move(fen, 'w'), &game) &&
         load_FEN(with_side_to_move(fen, 'b'), &game);
}

bool trace_fen(const std::string& fen, tuner_trace::row_trace_t* trace)
{
  if (!load_FEN(fen, &game)) { return false; }
  std::string error;
  return tuner_trace::trace_board(&game.board, trace, &error);
}

double e1_at(const tuner_trace::row_trace_t& trace, const double* shipped)
{
  dataset_t one;
  tuner_trace::append_row(trace, 0.5f, 0, 0, &one);
  return tuner_trace::smooth(tuner_trace::engine_model(), one, 0, shipped).e1;
}

bool eligible(const std::string& fen, const std::string& mirror, const double* p,
              tuner_trace::row_trace_t* a, tuner_trace::row_trace_t* b)
{
  if (!trace_fen(fen, a) || !trace_fen(mirror, b)) { return false; }
  return a->phase >= 1 && a->phase <= GAME_PHASE_MAX - 1 &&
         std::fabs(stage_two(*a, p)) < LAZY_EVAL_MARGIN - 1.0 &&
         std::fabs(stage_two(*b, p)) < LAZY_EVAL_MARGIN - 1.0 &&
         std::abs(a->engine_white) <= SCORE_LIMIT &&
         std::abs(b->engine_white) <= SCORE_LIMIT && loads_either_side(fen) &&
         loads_either_side(mirror) && e1_at(*a, p) != 0.0 && e1_at(*b, p) != 0.0;
}

void force(std::vector<double>* params)
{
  static const double cells[ENDGAME_SCALE_COUNT] = {64, 0, 24, 28, 68, 116, 96, 127, 120, 4, 40, 76, 92};
  static const double mopup[2][MOPUP_COUNT] = {{7, 11, 5}, {13, 9, 17}};
  for (size_t c = 0; c < ENDGAME_SCALE_COUNT; ++c) { (*params)[EGS_BASE + c] = cells[c]; }
  for (size_t i = 0; i < MOPUP_COUNT; ++i) {
    (*params)[MOP_MG_BASE + i] = mopup[0][i];
    (*params)[MOP_EG_BASE + i] = mopup[1][i];
  }
}

std::vector<double> labels(size_t n)
{
  std::vector<double> t(n);
  for (size_t i = 0; i < n; ++i) { t[i] = (i % 3 == 0) ? 1.0 : ((i % 3 == 1) ? 0.5 : 0.0); }
  return t;
}

std::vector<uint32_t> every(size_t n)
{
  std::vector<uint32_t> out(n);
  for (size_t i = 0; i < n; ++i) { out[i] = static_cast<uint32_t>(i); }
  return out;
}

// The case's check over its reached parameters, as the case writes it.
double normwise = 0.0;  // max |num - ana| over max |ana|, the reached parameters, DEC-354's (C)

void case_check(const dataset_t& data, std::vector<double> params, double* worst, size_t* at,
                double* second, size_t* second_at, std::vector<double>* analytic)
{
  double max_diff = 0.0, max_ana = 0.0;
  const std::vector<double> target = labels(data.size());
  const std::vector<uint32_t> index = every(data.size());
  tuner_trace::gradient(data, params.data(), 1.0, index, 0, data.size(), 1, target, analytic);
  std::vector<uint8_t> reached(PARAM_COUNT, 0);
  for (size_t p = 0; p < MATERIAL_COUNT; ++p) { reached[p] = 1; }
  for (const auto& f : tuner_trace::feature_table().features) {
    if (f.site == tuner_trace::SITE_POSITIONAL) { reached[f.eg] = 1; }
  }
  for (size_t p = MOP_MG_BASE; p < PARAM_COUNT; ++p) { reached[p] = 1; }
  const auto& model = tuner_trace::engine_model();
  *worst = 0; *second = 0; *at = 0; *second_at = 0;
  for (size_t p = 0; p < PARAM_COUNT; ++p) {
    if (!reached[p]) { continue; }
    const double saved = params[p];
    params[p] = saved + 0.01;
    const double up = tuner_trace::error_range(model, data, params.data(), 1.0, index, 0, index.size(), 1, target);
    params[p] = saved - 0.01;
    const double down = tuner_trace::error_range(model, data, params.data(), 1.0, index, 0, index.size(), 1, target);
    params[p] = saved;
    const double n = (up - down) / 0.02;
    max_diff = std::max(max_diff, std::fabs(n - (*analytic)[p]));
    max_ana = std::max(max_ana, std::fabs((*analytic)[p]));
    const double s = std::max(std::fabs(n), std::fabs((*analytic)[p]));
    const double e = (s > 1e-12) ? std::fabs(n - (*analytic)[p]) / s : std::fabs(n - (*analytic)[p]);
    if (e > *worst) { *second = *worst; *second_at = *at; *worst = e; *at = p; }
    else if (e > *second) { *second = e; *second_at = p; }
  }
  normwise = max_diff / max_ana;
}

TEST_CASE("option A")
{
  initialize_game_const_data(&game);
  std::vector<double> shipped(PARAM_COUNT, 0.0);
  starting_params(shipped.data());
  std::vector<double> params = shipped;
  force(&params);

  const std::vector<std::string> pop = test_endgame_positions::population();
  std::set<std::string> held(pop.begin(), pop.end());
  dataset_t data;
  for (const std::string& fen : pop) {
    tuner_trace::row_trace_t t;
    REQUIRE(trace_fen(fen, &t));
    if (e1_at(t, shipped.data()) == 0.0) { continue; }
    tuner_trace::append_row(t, 0.5f, 0, 0, &data);
  }
  double worst, second;
  size_t at, second_at;
  std::vector<double> analytic;
  case_check(data, params, &worst, &at, &second, &second_at, &analytic);
  printf("base: %zu rows; the case's worst %.6g at %zu, next %.6g at %zu; normwise %.4g\n", data.size(), worst, at,
         second, second_at, normwise);

  const std::vector<std::string> candidates = test_endgame_positions::population(80);
  size_t next = 0;
  std::set<std::string> tried;
  std::vector<std::string> picked;
  for (int round = 0; round < 3 && worst >= 1e-6; ++round) {
    const size_t target_param = at;
    const double base_g = analytic[target_param];
    printf("round %d: TARGET %zu, base |g| %.6e, the rule wants %.6e\n", round + 1, target_param,
           std::fabs(base_g), 3.0 * std::fabs(base_g));
    bool found = false;
    size_t eligible_tried = 0;
    for (; next < candidates.size(); ++next) {
      const std::string& fen = candidates[next];
      const std::string mirror = mirror_fen(fen);
      if (held.count(fen) || held.count(mirror) || tried.count(fen) || tried.count(mirror)) { continue; }
      tried.insert(fen);
      tuner_trace::row_trace_t a, b;
      if (!eligible(fen, mirror, shipped.data(), &a, &b)) { continue; }
      eligible_tried++;
      dataset_t with = data;
      tuner_trace::append_row(a, 0.5f, 0, 0, &with);
      tuner_trace::append_row(b, 0.5f, 0, 0, &with);
      std::vector<double> g;
      tuner_trace::gradient(with, params.data(), 1.0, every(with.size()), 0, with.size(), 1,
                            labels(with.size()), &g);
      if (std::fabs(g[target_param]) >= 3.0 * std::fabs(base_g)) {
        printf("  pick after %zu eligible pairs (candidate %zu of %zu): |g| %.6e\n    %s\n    %s\n",
               eligible_tried, next + 1, candidates.size(), std::fabs(g[target_param]), fen.c_str(),
               mirror.c_str());
        data = with;
        picked.push_back(fen);
        picked.push_back(mirror);
        held.insert(fen);
        held.insert(mirror);
        found = true;
        ++next;
        break;
      }
    }
    if (!found) {
      printf("  no eligible pair qualifies (%zu eligible tried)\n", eligible_tried);
      break;
    }
    case_check(data, params, &worst, &at, &second, &second_at, &analytic);
    printf("  the case with %zu rows: worst %.6g at %zu (|g| %.3e), next %.6g at %zu; normwise %.4g\n", data.size(),
           worst, at, std::fabs(analytic[at]), second, second_at, normwise);
  }
  printf("picked %zu rows\n", picked.size());
  for (const std::string& f : picked) { printf("PICK %s\n", f.c_str()); }
}
