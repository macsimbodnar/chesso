bool write_rows(const std::string& path, const std::vector<std::string>& fens)
{
  std::ofstream out(path);
  if (!out) { return false; }
  static const double results[3] = {1.0, 0.5, 0.0};
  int score = -280;
  for (size_t i = 0; i < fens.size(); ++i) {
    const int phase = engine_phase(fens[i]);
    if (phase < 0) { return false; }
    out << fens[i] << '\t' << results[i % 3] << '\t' << score << '\t'
        << phase % (GAME_PHASE_MAX + 1) << '\n';
    score += 41;
    if (score > 300) { score -= 600; }
  }
  return out.good();
}

  const bool ok = tuner_trace::load(path, data, &report, &error, 1) && report.mismatches == 0;
  std::remove(path.c_str());
  return ok;
}

// S083_tuner_preconditions.cpp's stage_two() as S122's phase 2 fixed it: the curve included.
double stage_two(const dataset_t& data, size_t row, const double* p)
{
  const std::vector<tuner_trace::feature_t>& features = tuner_trace::feature_table().features;
  const double mg_weight = data.phase[row] / static_cast<double>(GAME_PHASE_MAX);
  const double eg_weight = 1.0 - mg_weight;
  double sum = 0.0;
  for (uint32_t j = data.offsets[row]; j < data.offsets[row + 1]; ++j) {
    const tuner_trace::entry_t e = data.entries[j];
    const tuner_trace::feature_t& f = features[e.feature];
    if (!tuner_trace::clamped_site(f.site)) { continue; }
    sum += e.coef * (p[f.mg] * mg_weight + p[f.eg] * eg_weight);
  }
  double x[2][2] = {{0.0, 0.0}, {0.0, 0.0}};
  for (uint32_t j = data.side_offsets[row]; j < data.side_offsets[row + 1]; ++j) {
    const tuner_trace::side_entry_t e = data.side_entries[j];
    if (e.group != 0) { continue; }
    x[e.side][0] += e.coef * p[KSC_MG_BASE + e.input];
    x[e.side][1] += e.coef * p[KSC_EG_BASE + e.input];
  }
  for (int side = 0; side < 2; ++side) {
    const double own = (side == 0) ? -1.0 : 1.0;
    const double m = std::max(x[side][0], 0.0);
    const double n = std::max(x[side][1], 0.0);
    sum += own * (m * m * mg_weight + n * n * eg_weight) / KS_CURVE_DIVISOR;
  }
  return sum;
}

struct eval_t
{
  std::vector<uint32_t> index;
  std::vector<double> target;
};

eval_t every(const dataset_t& data)
{
  eval_t e;
  for (size_t i = 0; i < data.size(); ++i) {
    e.index.push_back(static_cast<uint32_t>(i));
    e.target.push_back(data.result[i]);
  }
  return e;
}

std::vector<double> analytic_of(const dataset_t& data, const std::vector<double>& params,
                                const std::vector<uint32_t>& index, const std::vector<double>& target)
{
  std::vector<double> g;
  tuner_trace::gradient(data, params.data(), 1.0, index, 0, index.size(), 1, target, &g);
  return g;
}

std::vector<double> numeric_of(const dataset_t& data, std::vector<double> params, double h,
                               const eval_t& e)
{
  const tuner_trace::model_t& model = tuner_trace::engine_model();
  std::vector<double> d(model.param_count, 0.0);
  for (size_t p = 0; p < model.param_count; ++p) {
    const double saved = params[p];
    params[p] = saved + h;
    const double up = tuner_trace::error_range(model, data, params.data(), 1.0, e.index, 0,
                                               e.index.size(), 1, e.target);
    params[p] = saved - h;
    const double down = tuner_trace::error_range(model, data, params.data(), 1.0, e.index, 0,
                                                 e.index.size(), 1, e.target);
    params[p] = saved;
    d[p] = (up - down) / (2.0 * h);
  }
  return d;
}

double rel(double num, double ana)
{
  const double scale = std::max(std::fabs(num), std::fabs(ana));
  const double diff = std::fabs(num - ana);
  return (scale > 1e-12) ? diff / scale : diff;
}

struct worst_t { double value = 0.0; size_t param = 0; };

worst_t worst_rel(const std::vector<double>& num, const std::vector<double>& ana)
{
  worst_t w;
  for (size_t p = 0; p < num.size(); ++p) {
    const double r = rel(num[p], ana[p]);
    if (r > w.value) { w = {r, p}; }
  }
  return w;
}

double normwise(const std::vector<double>& num, const std::vector<double>& ana)
{
  double diff = 0.0, top = 0.0;
  for (size_t p = 0; p < num.size(); ++p) {
    diff = std::max(diff, std::fabs(num[p] - ana[p]));
    top = std::max(top, std::fabs(ana[p]));
  }
  return diff / top;
}

std::vector<double> pushed(const std::vector<double>& start, int push)
{
  std::vector<double> params = start;
  for (size_t t = 0; t < KING_SAFETY_COUNT; ++t) {
    params[KS_MG_BASE + t] += push;
    params[KS_EG_BASE + t] += push;
  }
  return params;
}

struct row_t
{
  int push = 0;
  size_t bound = 0;
  double bound_at = 0.0, next = 0.0, kink = 1e300;
  double clamped = -1.0, rest = -1.0;
  bool binds = false, clear = false, silent = false;
  worst_t r005, r0025, rich;
  double norm005 = 0.0;
  std::vector<double> ana, d005, d0025;
};

row_t read_push(const dataset_t& data, const std::vector<double>& start, int push)
{
  const double margin = static_cast<double>(LAZY_EVAL_MARGIN);
  row_t r;
  r.push = push;
  const std::vector<double> params = pushed(start, push);
  std::vector<uint32_t> binding;
  for (size_t row = 0; row < data.size(); ++row) {
    const double two = std::fabs(stage_two(data, row, params.data()));
    r.kink = std::min(r.kink, std::fabs(two - margin));
    if (two > margin) {
      binding.push_back(static_cast<uint32_t>(row));
      r.bound_at = std::max(r.bound_at, two);
    } else {
      r.next = std::max(r.next, two);
    }
  }
  r.bound = binding.size();
  const eval_t e = every(data);
  r.ana = analytic_of(data, params, e.index, e.target);
  r.d005 = numeric_of(data, params, 0.005, e);
  r.d0025 = numeric_of(data, params, 0.0025, e);
  std::vector<double> rich(r.d005.size());
  for (size_t p = 0; p < rich.size(); ++p) { rich[p] = (4.0 * r.d0025[p] - r.d005[p]) / 3.0; }
  r.r005 = worst_rel(r.d005, r.ana);
  r.r0025 = worst_rel(r.d0025, r.ana);
  r.rich = worst_rel(rich, r.ana);
  r.norm005 = normwise(r.d005, r.ana);
  if (!binding.empty()) {
    std::vector<double> target(data.size());
    for (size_t i = 0; i < data.size(); ++i) { target[i] = data.result[i]; }
    std::vector<double> on;
    tuner_trace::gradient(data, params.data(), 1.0, binding, 0, binding.size(), 1, target, &on);
    std::vector<uint8_t> clamped(PARAM_COUNT, 0);
    for (const tuner_trace::feature_t& f : tuner_trace::feature_table().features) {
      if (tuner_trace::clamped_site(f.site)) { clamped[f.mg] = 1; clamped[f.eg] = 1; }
    }
    for (const auto& [mg, eg] : tuner_trace::engine_model().groups[0].inputs) {
      clamped[mg] = 1;
      clamped[eg] = 1;
    }
    r.clamped = 0.0;
    r.rest = 0.0;
    for (size_t p = 0; p < PARAM_COUNT; ++p) {
      if (clamped[p]) { r.clamped = std::max(r.clamped, std::fabs(on[p])); }
      else { r.rest = std::max(r.rest, std::fabs(on[p])); }
    }
  }
  r.binds = !binding.empty() && binding.size() < data.size();
  r.clear = r.kink > 1.0;
  r.silent = r.binds && r.clamped == 0.0 && r.rest > 1e-9;
  return r;
}

struct option_t
{
  const char* name;
  std::function<double(const row_t&)> metric;
  double limit;
};

// The rule: of the pushes meeting all four criteria, the farthest nearest row from the kink,
// ties to the smaller push. -1 when none.
int pick(const std::vector<row_t>& rows, const option_t& o)
{
  int chosen = -1;
  double best = -1.0;
  for (const row_t& r : rows) {
    if (r.binds && r.clear && r.silent && o.metric(r) <= o.limit && r.kink > best) {
      chosen = static_cast<int>(&r - rows.data());
      best = r.kink;
    }
  }
  return chosen;
}

void print_table(const std::vector<row_t>& rows)
{
  printf(" push bound  bound at     next  kink dist  rel h.005 (param)   rel h.0025  normwise  Richardson (param)  clamped     rest\n");
  for (const row_t& r : rows) {
    printf("%+5d %5zu %9.1f %8.1f %10.2f %10.3g (%4zu) %12.3g %9.3g %10.3g (%4zu) %8.3g %8.3g%s%s%s\n", r.push, r.bound,
           r.bound_at, r.next, r.kink, r.r005.value, r.r005.param, r.r0025.value, r.norm005, r.rich.value,
           r.rich.param, r.clamped, r.rest, r.binds ? "" : "  binds none/all", r.clear ? "" : "  kink<=1",
           (r.binds && !r.silent) ? "  not silent" : "");
  }
}

void five_worst(const row_t& r, const char* label)
{
  std::vector<size_t> order(r.ana.size());
  for (size_t p = 0; p < order.size(); ++p) { order[p] = p; }
  std::sort(order.begin(), order.end(), [&](size_t a, size_t b) {
    return rel(r.d005[a], r.ana[a]) > rel(r.d005[b], r.ana[b]);
  });
  printf("== %s, push %+d: the five worst parameters by relative error at h 0.005\n", label, r.push);
  printf("  param name                       relative        analytic   numeric h.005  |num - ana|  rel h.0025  Richardson\n");
  for (size_t i = 0; i < 5; ++i) {
    const size_t p = order[i];
    const double richardson = (4.0 * r.d0025[p] - r.d005[p]) / 3.0;
    printf("  %5zu %-24s %9.3e %+15.8e %+15.8e %11.3e %11.3e %11.3e\n", p, param_name(p).c_str(),
           rel(r.d005[p], r.ana[p]), r.ana[p], r.d005[p], std::fabs(r.d005[p] - r.ana[p]),
           rel(r.d0025[p], r.ana[p]), rel(richardson, r.ana[p]));
  }
}

void report_options(const std::vector<row_t>& rows, const std::vector<option_t>& options)
{
  for (const option_t& o : options) {
    const int i = pick(rows, o);
    if (i < 0) {
      printf("%-48s none\n", o.name);
      continue;
    }
    const row_t& r = rows[i];
    const double m = o.metric(r);
    printf("%-48s %+d: %zu rows bound, kink %.2f, metric %.3g (rel h.005 %.3g at %zu %s), limit %.1e, margin x%.2f\n",
           o.name, r.push, r.bound, r.kink, m, r.r005.value, r.r005.param, param_name(r.r005.param).c_str(),
           o.limit, o.limit / m);
  }
}

// The fixture-level readings DEC-316's rows can move: the gradient case's worst (shipped
// constants, h 0.01, every parameter) and the largest |stage two| at the shipped constants.
void fixture_readings(const dataset_t& data, const std::vector<double>& start, const char* label)
{
  const eval_t e = every(data);
  const std::vector<double> ana = analytic_of(data, start, e.index, e.target);
  const worst_t w = worst_rel(numeric_of(data, start, 0.01, e), ana);
  double clear = 0.0;
  for (size_t row = 0; row < data.size(); ++row) {
    clear = std::max(clear, std::fabs(stage_two(data, row, start.data())));
  }
  printf("%s: %zu rows; the gradient case's worst %.3e at %zu %s; the fixture's largest |stage two| %.1f\n", label,
         data.size(), w.value, w.param, param_name(w.param).c_str(), clear);
}

double stage_two_of(const std::string& fen, const std::vector<double>& start, int* engine_white, int* phase)
{
  if (!load_FEN(fen, &game)) { return 1e300; }
  tuner_trace::row_trace_t trace;
  std::string error;
  if (!tuner_trace::trace_board(&game.board, &trace, &error)) { return 1e300; }
  *engine_white = trace.engine_white;
  *phase = trace.phase;
  dataset_t one;
  if (!load_rows({fen}, &one)) { return 1e300; }
  return stage_two(one, 0, start.data());
}

std::string with_side_to_move(const std::string& fen, char side)
{
  std::string flipped = fen;
  flipped[fen.find(' ') + 1] = side;
  return flipped;
}

bool loads_either_side(const std::string& fen)
{
  return load_FEN(with_side_to_move(fen, 'w'), &game) && load_FEN(with_side_to_move(fen, 'b'), &game);
}

std::string canonical_castling(const std::string& fen)
{
  std::vector<std::string> parts = split_string(fen);
  if (parts.size() != 6 || parts[2] == "-") { return fen; }
  std::string rights;
  for (const char c : std::string("KQkq")) {
    if (parts[2].find(c) != std::string::npos) { rights += c; }
  }
  parts[2] = rights;
  std::string out;
  for (size_t i = 0; i < parts.size(); ++i) { out += (i ? " " : "") + parts[i]; }
  return out;
}

bool eligible(const std::string& fen, const std::string& mirror, const std::vector<double>& start)
{
  int ew = 0, em = 0, pw = 0, pm = 0;
  const double a = stage_two_of(fen, start, &ew, &pw);
  const double b = stage_two_of(mirror, start, &em, &pm);
  const double margin = static_cast<double>(LAZY_EVAL_MARGIN);
  return a < 1e299 && b < 1e299 && pw >= 1 && pw <= GAME_PHASE_MAX - 1 && std::fabs(a) < margin - 1.0 &&
         std::fabs(b) < margin - 1.0 && std::abs(ew) <= 800 && std::abs(em) <= 800 && loads_either_side(fen) &&
         loads_either_side(mirror);
}

// S101_gradient_rows.cpp's rule for TARGET over ROWS: the first eligible pair of all_test_fens()
// that puts |dE/d TARGET| at the shipped constants at or above 3 times the list's.
bool pick_pair(const std::vector<std::string>& rows, size_t target, const std::vector<double>& start,
               std::string* fen_out, std::string* mirror_out)
{
  dataset_t base;
  if (!load_rows(rows, &base)) { return false; }
  const eval_t eb = every(base);
  const double g0 = std::fabs(analytic_of(base, start, eb.index, eb.target)[target]);
  printf("pair for %zu %s: base |g| %.6e over %zu rows, the rule wants %.6e\n", target, param_name(target).c_str(), g0,
         rows.size(), 3.0 * g0);
  size_t tried = 0;
  for (const std::string& fen : all_test_fens()) {
    const std::string mirror = canonical_castling(mirror_fen(fen));
    bool known = false;
    for (const std::string& row : rows) {
      if (row == fen || row == mirror) { known = true; }
    }
    if (known || !eligible(fen, mirror, start)) { continue; }
    std::vector<std::string> with = rows;
    with.push_back(fen);
    with.push_back(mirror);
    dataset_t data;
    if (!load_rows(with, &data)) {
      printf("  an eligible pair failed to load: %s\n", fen.c_str());
      return false;
    }
    tried++;
    const eval_t e = every(data);
    const double g = std::fabs(analytic_of(data, start, e.index, e.target)[target]);
    if (g >= 3.0 * g0) {
      printf("  pick after %zu eligible pairs: |g| %.6e\n    \"%s\",\n    \"%s\",\n", tried, g, fen.c_str(),
             mirror.c_str());
      *fen_out = fen;
      *mirror_out = mirror;
      return true;
    }
  }
  printf("  none of %zu eligible pairs triples it\n", tried);
  return false;
}

std::vector<row_t> sweep(const dataset_t& data, const std::vector<double>& start, int lo, int hi, int step)
{
  std::vector<row_t> rows;
  for (int push = lo; push <= hi; push += step) { rows.push_back(read_push(data, start, push)); }
  return rows;
}

TEST_CASE("S122 clamp reading")
{
  initialize_game_const_data(&game);
  std::vector<double> start(PARAM_COUNT, 0.0);
  starting_params(start.data());

  const std::vector<option_t> options = {
      {"rule as written (rel h.005 <= 2e-7)", [](const row_t& r) { return r.r005.value; }, 2e-7},
      {"(A) rel h.005 <= 4e-7", [](const row_t& r) { return r.r005.value; }, 4e-7},
      {"(B) rel h.0025 <= 2e-7", [](const row_t& r) { return r.r0025.value; }, 2e-7},
      {"(C) normwise h.005 <= 2e-7", [](const row_t& r) { return r.norm005; }, 2e-7},
      {"(D) rel h.005 <= 1e-6", [](const row_t& r) { return r.r005.value; }, 1e-6},
      {"(F) Richardson from h.005 <= 2e-7", [](const row_t& r) { return r.rich.value; }, 2e-7},
  };

  std::vector<std::string> rows = test_eval_positions::positions;
  dataset_t data;
  REQUIRE(load_rows(rows, &data));
  printf("clamp reading: %zu fixture rows, LAZY_EVAL_MARGIN %d, K 1; pushes +10 to +200 by 5\n", data.size(),
         static_cast<int>(LAZY_EVAL_MARGIN));
  fixture_readings(data, start, "the fixture as staged");
  const std::vector<row_t> table = sweep(data, start, 10, 200, 5);
  print_table(table);
  printf("\n== the rule's pick by option\n");
  report_options(table, options);
  for (const char* which : {"(A) rel h.005 <= 4e-7", "(D) rel h.005 <= 1e-6"}) {
    for (const option_t& o : options) {
      if (std::string(o.name) != which) { continue; }
      const int i = pick(table, o);
      if (i >= 0) { five_worst(table[i], which); }
    }
  }

  // (E): DEC-316's rows for 984 and 996, then for whatever is worst, at most three pairs.
  printf("\n== (E) DEC-316's rows, S101_gradient_rows.cpp's rule, rel h.005 <= 2e-7\n");
  const option_t e_option = options[0];
  std::vector<size_t> targets = {PKO_MG_BASE + 5, PKE_MG_BASE + 5};
  std::vector<row_t> e_table;
  for (int pair = 0; pair < 3; ++pair) {
    size_t target;
    if (pair < 2) {
      target = targets[pair];
    } else {
      // The weight worst at the push nearest passing: binding, clear of the kink, silent,
      // the smallest rel h.005.
      const row_t* best = nullptr;
      for (const row_t& r : e_table) {
        if (r.binds && r.clear && r.silent && (!best || r.r005.value < best->r005.value)) { best = &r; }
      }
      if (!best) { printf("no push binds clear and silent; stop\n"); break; }
      target = best->r005.param;
      printf("third target: %zu %s, worst at %+d (%.3g)\n", target, param_name(target).c_str(), best->push,
             best->r005.value);
    }
    std::string fen, mirror;
    if (!pick_pair(rows, target, start, &fen, &mirror)) { break; }
    rows.push_back(fen);
    rows.push_back(mirror);
    dataset_t with;
    REQUIRE(load_rows(rows, &with));
    char label[64];
    snprintf(label, sizeof label, "with %d pair(s)", pair + 1);
    fixture_readings(with, start, label);
    e_table = sweep(with, start, 10, 200, 5);
    print_table(e_table);
    const int i = pick(e_table, e_option);
    if (i >= 0) {
      printf("(E) with %d pair(s): +%d passes, %zu rows bound, kink %.2f, rel h.005 %.3g at %zu %s, margin x%.2f\n",
             pair + 1, e_table[i].push, e_table[i].bound, e_table[i].kink, e_table[i].r005.value,
             e_table[i].r005.param, param_name(e_table[i].r005.param).c_str(), 2e-7 / e_table[i].r005.value);
      five_worst(e_table[i], "(E)");
      break;
    }
    printf("(E) with %d pair(s): no push passes\n", pair + 1);
  }
}
