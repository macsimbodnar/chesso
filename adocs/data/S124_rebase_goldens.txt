// Scratch probe, never delivered: over a corpus (FEN column), each row's smooth-exact
// gap against two per-row curve allowances at the library's weights: L, +1 where any
// king's x is > 0 in either half; E, the curve's own residual |d_mg phase + d_eg (24 -
// phase)| / 24, d_h the two kings' T fractions, White's taken off.
#define DOCTEST_CONFIG_IMPLEMENT_WITH_MAIN
#include <doctest.h>
#include <cmath>
#include <cstdio>
#include <fstream>
#include <string>
#include <vector>
#include "bitboard.hpp"
#include "tuner_layout.hpp"
#include "tuner_trace.hpp"

TEST_CASE("curveprobe")
{
  game_t game;
  initialize_game_const_data(&game);
  std::ifstream in(std::getenv("CP_LIST"));
  std::vector<double> params(tuner_layout::PARAM_COUNT, 0.0);
  tuner_layout::starting_params(params.data());
  const std::vector<int64_t> w = tuner_trace::engine_weights();
  std::string line;
  size_t rows = 0, fires = 0, e_zero = 0, viol_e = 0, viol_l = 0, near_e = 0, near_l = 0;
  size_t cell_past_l = 0, cell_past_e = 0, cells = 0;
  double worst_quiet = 0, worst_slack_e = -9, worst_cell_over_e = -9, worst_cell_over_l = -9;
  std::string fen_slack, fen_cell_e;
  while (std::getline(in, line)) {
    const std::string fen = line.substr(0, line.find('\t'));
    if (!load_FEN(fen, &game)) { continue; }
    tuner_trace::row_trace_t trace;
    std::string error;
    if (!tuner_trace::trace_board(&game.board, &trace, &error)) { continue; }
    tuner_trace::dataset_t data;
    tuner_trace::append_row(trace, 0.5f, 0, 0, &data);
    const double gap = std::fabs(tuner_trace::score(data, 0, params.data()) -
                                 static_cast<double>(tuner_trace::exact_score(data, 0, w.data())));
    const tuner_trace::exact_parts_t parts = tuner_trace::exact_parts(data, 0, w.data());
    int64_t x[2][2] = {{0, 0}, {0, 0}};
    for (const tuner_trace::side_entry_t& e : data.side_entries) {
      if (e.group != 0) { continue; }
      x[e.side][0] += e.coef * w[tuner_layout::KSC_MG_BASE + e.input];
      x[e.side][1] += e.coef * w[tuner_layout::KSC_EG_BASE + e.input];
    }
    bool fire = false;
    double d[2] = {0, 0};
    for (int s = 0; s < 2; ++s) {
      for (int h = 0; h < 2; ++h) {
        if (x[s][h] > 0) { fire = true; }
        const int64_t p = std::max<int64_t>(x[s][h], 0);
        const double frac = static_cast<double>((p * p) % KS_CURVE_DIVISOR) / KS_CURVE_DIVISOR;
        d[h] += (s == 0 ? -1.0 : 1.0) * frac;
      }
    }
    const int phase = data.phase[0];
    const double a = std::fabs(d[0] * phase + d[1] * (24 - phase)) / 24.0;
    const bool off_f = parts.cell != eval_trace::NO_CELL &&
                       w[tuner_layout::EGS_BASE + parts.cell] != ENDGAME_SCALE_ONE;
    const double base = off_f ? 2.0 * 23.0 / 24.0 + 3071.0 / 3072.0 : 69.0 / 24.0;
    const double bl = base + (fire ? 1.0 : 0.0), be = base + a;
    rows++;
    fires += fire;
    e_zero += (a == 0.0);
    if (gap > bl + 1e-9) { viol_l++; }
    if (gap > be + 1e-9) { viol_e++; }
    if (gap > bl - 0.03) { near_l++; }
    if (gap > be - 0.03) { near_e++; }
    if (!fire) { worst_quiet = std::max(worst_quiet, gap); }
    if (gap - be > worst_slack_e) { worst_slack_e = gap - be; fen_slack = fen; }
    if (off_f) {
      cells++;
      const double at_f_l = 69.0 / 24.0 + (fire ? 1.0 : 0.0), at_f_e = 69.0 / 24.0 + a;
      if (gap > at_f_l) { cell_past_l++; }
      if (gap > at_f_e) { cell_past_e++; }
      worst_cell_over_l = std::max(worst_cell_over_l, gap - at_f_l);
      if (gap - at_f_e > worst_cell_over_e) { worst_cell_over_e = gap - at_f_e; fen_cell_e = fen; }
    }
  }
  printf("rows %zu; L fires on %zu; E allowance zero on %zu\n", rows, fires, e_zero);
  printf("bound violations: L %zu, E %zu; rows within 0.03 of the bound: L %zu, E %zu\n", viol_l, viol_e, near_l, near_e);
  printf("worst gap on rows L does not fire on %.6f; worst gap - E bound %.6f at %s\n", worst_quiet, worst_slack_e, fen_slack.c_str());
  printf("rows off F %zu; past a row at F's bound: L %zu (worst over %.6f), E %zu (worst over %.6f at %s)\n", cells,
         cell_past_l, worst_cell_over_l, cell_past_e, worst_cell_over_e, fen_cell_e.c_str());
}
